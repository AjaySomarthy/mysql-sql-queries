3Q : Write a SQL code for Window Functions (row_number, rank and dense_rank) ?

Ans :
-- 1) creating Students table
create table Students (
  name varchar(255),
marks int);

-- adding values into the table
insert into Students (name, marks)
values
("Anusha", 95),
("Bindhu", 90),
("Chitra", 90),
("Divya", 85)

-- RowNumber, Rank and DenseRank code
select name, marks,
row_number() over( order by (marks) desc) as RowNumber,
rank() over( order by (marks) desc) as Rank,
dense_rank() over( order by (marks) desc) as DenseRank
from Students;

-- RowNumber, Rank and DenseRank table
name    marks  RowNumber  Rank	 DenseRank

Bindhu	 90	 2	   2	   2

4Q : Write a code for finding the Second highest and third highest marks ?

Ans :
-- 1) creating Students table
create table Students (
  name varchar(255),
marks int);

-- adding values into the table
insert into Students (name, marks)
values
("Anusha", 95),
("Bindhu", 90),
("Chitra", 90),
("Divya", 85);

-- second highest marks code
select name, marks
from
(select name, marks,
dense_rank() over (order by (marks) desc) as DenseRank
 from Students)
where DenseRank=2;

-- output :
name	marks
Bindhu	 90
Chitra	 90

-- third highest marks code
select name, marks
from
(select name, marks,
dense_rank() over (order by (marks) desc) as DenseRank
 from Students)
where DenseRank=3;

-- output :
name	marks
Divya	 85

7Q : Write a SQL query for CASE expression?

Case Expression :
The CASE expression is used to define different results based on specified conditions in an SQL statement.
The CASE expression goes through the conditions and stops at the first match (like an if-then-else statement).
So, once a condition is true, it will stop processing and return the result.
If no conditions are true, it returns the value in the ELSE clause. If there is no ELSE clause and no conditions are true, it returns NULL.

-- input table : Customers
customer_id	first_name	last_name	age	country
1	           John	           Doe	        31	USA
2	           Robert	   Luna	        22	USA
3	           David	   Robinson	22	UK
4	           John	           Reinhardt	25	UK
5	           Betty	   Doe   	28	UAE

-- case expression 1 :
select customer_id, first_name, age,
case
when age = 25 then 'Major'
when age > 25 then 'Adult'
else 'Minor'
end as AgeStatus
from Customers;

-- output table 1 :
customer_id	first_name	age    AgeStatus
1	         John	        31	Adult
2	         Robert	        22	Minor
3	         David	        22	Minor
4	         John	        25	Major
5	         Betty	        28	Adult

-- case expression 2 :
select *,
case
when age >= 25 then 'Major'
else 'Minor'
end as Age_Status
from Customers;

-- output table 2 :
customer_id	first_name	last_name	age	country   Age_Status
1	           John	           Doe	        31	USA       Major
2	           Robert	   Luna	        22	USA       Minor
3	           David	   Robinson	22	UK        Minor
4	           John	           Reinhardt	25	UK        Major
5	           Betty	   Doe   	28	UAE       Major

***2Q : 1) Create a StudentsDepartment table with columns as StudentID, StudentName, DepartmentName and studentMarks.
     2) Insert any random values into it.
     3) Find the RowNumber, Rank and DenseRank from each department wise.
     4) Find the Second highest rank from each department wise.
     5) Find the third highest rank from each department wise.
     6) Find the sum, avg, min, max of student marks from each department.
     7) Find the sum, avg, min, max, diff(max,min) of student marks from each department.
     7) Find the sum, avg, min, max, diff(max,min) of 1st greatest and 3rd greatest student marks from each department.
     7) Find the sum, avg, min, max, diff(max,min) of greatest two student marks from each department.

Ans : 1) create table StudentsDepartment (
         StudentID int,
         StudentName varchar(255),
         DepartmentName varchar(255),
         StudentMarks int );

      2) insert into StudentsDepartment (StudentID, StudentName, DepartmentName, StudentMarks)
         values (101,"Abhi","ECE",99), (102,"Ajay","ECE",95), (103,"Akhil","ECE",95), (104,"Arjun","ECE",90),
         (201,"Balu","CSE",90),(202,"Bharath","CSE",90),(203,"Bhuvi","CSE",90),(204,"Bindhu","CSE",80),(205,"Bittu","CSE",70);

      3) select StudentID, StudentName, DepartmentName, StudentMarks,
         row_number() over(partition by DepartmentName order by StudentMarks desc) as RowNumber,
         rank() over(partition by DepartmentName order by StudentMarks desc) as Rank,
         dense_rank() over(partition by DepartmentName order by StudentMarks desc) as DenseRank
         from StudentsDepartment;

      4) select StudentID, StudentName, DepartmentName, StudentMarks from
            (select StudentID, StudentName, DepartmentName, StudentMarks,
            dense_rank() over(partition by DepartmentName order by StudentMarks desc) as DenseRank
            from StudentsDepartment)
         where DenseRank = 2;

      5) select StudentID, StudentName, DepartmentName, StudentMarks from
            (select StudentID, StudentName, DepartmentName, StudentMarks,
            dense_rank() over(partition by DepartmentName order by StudentMarks desc) as DenseRank
            from StudentsDepartment)
         where DenseRank = 3;

      6) select DepartmentName,
         sum(StudentMarks) as TotalMarks,
         avg(StudentMarks) as AvgMarks, min(StudentMarks) as MinMarks,
         max(StudentMarks) as MaxMarks
         from StudentsDepartment
         group by DepartmentName
         order by DepartmentName;

7) first approach :
select DepartmentName, TotalMarks, AvgMarks, MinMarks, MaxMarks,
MaxMarks-MinMarks as DiffMarks from
(select DepartmentName,
sum(StudentMarks) as TotalMarks,
avg(StudentMarks) as AvgMarks,
min(StudentMarks) as MinMarks,
max(StudentMarks) as MaxMarks
from StudentsDepartment
group by DepartmentName
order by DepartmentName desc)

7) second approach :
with CTE as
(select DepartmentName,
sum(StudentMarks) as TotalMarks,
avg(StudentMarks) as AvgMarks,
min(StudentMarks) as MinMarks,
max(StudentMarks) as MaxMarks
from StudentsDepartment
group by DepartmentName
order by DepartmentName desc)
select DepartmentName, TotalMarks, AvgMarks, MinMarks, MaxMarks,
MaxMarks-MinMarks as DiffMarks from CTE

8)
with CTE as

 select DepartmentName,
 sum(StudentMarks) as TotalMarks,
 avg(StudentMarks) as AvgMarks,
 min(StudentMarks) as MinMarks,

 max(StudentMarks)-min(StudentMarks) as DiffMarks

9)
with CTE as
(select  StudentID, StudentName, DepartmentName, StudentMarks, DenseRank from
